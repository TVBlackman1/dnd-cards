\newcommand{\SpellComponents}[3]{%
  \def\Comp{}%
  \DTLifstringeq{#1}{TRUE}{\def\Comp{В}}{}%
  \DTLifstringeq{#2}{TRUE}{%
    \ifx\Comp\empty
      \def\Comp{С}%
    \else
      \edef\Comp{\Comp,С}%
    \fi
  }{}%
  \DTLifstringgt{#3}{}{%
    \ifx\Comp\empty
      \def\Comp{М}%
    \else
      \edef\Comp{\Comp,М (#3)}%
    \fi
  }{}%
  \Comp
}
