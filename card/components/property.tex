\usepackage{calc} % для \widthof

\newsavebox{\propbox}

\newcommand{\SpellProperty}[2]{%
  \noindent
  \sbox{\propbox}{
    % --- icon minipage ---
    \begin{minipage}[c]{\widthof{\includegraphics[width=3mm]{#1}}}
      \includegraphics[width=3mm]{#1}%
    \end{minipage}%
    \hspace{4pt}% gap между иконкой и текстом
    % --- text minipage ---
    \begin{minipage}[c]{\widthof{\textbf{\fontsize{7pt}{12}\selectfont #2}}}
      \textbf{\fontsize{7pt}{12}\selectfont #2}%
    \end{minipage}%
  }

  \begin{tcolorbox}[
    sharp corners,
    boxrule=0pt,
    enhanced jigsaw,
    colback=red!20!yellow,
    opacityframe=1,opacityback=0,
    left=0pt, right=0pt, top=0pt, bottom=0pt,
    before skip=0pt, after skip=0pt,
    boxsep=0pt,
    nobeforeafter,
    width=\wd\propbox,
  ]
    \usebox{\propbox}
  \end{tcolorbox}%
}
