\newsavebox{\testbox}
\newcommand{\autoCard}[1]{%
  \savebox{\testbox}{\begin{minipage}{4cm}#1\end{minipage}}%
  \ifdim\ht\testbox<0.3cm
    \begin{cardS}#1\end{cardS}%
  \else
    \savebox{\testbox}{\begin{minipage}{6cm}#1\end{minipage}}%
    \ifdim\ht\testbox<0.4cm
      \begin{cardM}#1\end{cardM}%
    \else
      \begin{cardL}#1\end{cardL}%
    \fi
  \fi
}
