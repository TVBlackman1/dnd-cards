\newcommand{\SpellDescrition}[1]{%
  \begingroup
    \setlength{\parindent}{0pt}
    \setlength{\parskip}{2mm}
    \fontsizeBody\fontsize{8pt}{8pt}\fontspec{EB Garamond}\selectfont
    \leftskip=4pt
    \rightskip=4pt
    \begingroup
      \catcode`\\=\active
      \def\\{\par}
      #1
    \endgroup
    \par
  \endgroup
}

\newsavebox{\testbox}
\newcommand{\autoCard}[1]{%
  \savebox{\testbox}{\begin{minipage}{4cm}#1\end{minipage}}%
  \ifdim\ht\testbox<0.3cm
    \begin{cardS}#1\end{cardS}%
  \else
    \savebox{\testbox}{\begin{minipage}{6cm}#1\end{minipage}}%
    \ifdim\ht\testbox<0.4cm
      \begin{cardM}#1\end{cardM}%
    \else
      \begin{cardL}#1\end{cardL}%
    \fi
  \fi
}

\DTLforeach*{spells}{
  \spellName=Name,
  \spellSchool=School,
  \spellLevel=Level,
  \spellCastTime=CastTime,
  \spellDuration=Duration,
  \spellDistance=Distance,
  \spellVerbal=Verbal,
  \spellSomatic=Somatic,
  \spellMaterial=Material,
  \spellDescription=Description%
}{
    % \autoCard{\spellSchool}
    \resolveCardColor{\spellSchool}
    \begin{tcolorbox}[
      spellcard, title=\spellName,
      after title=\hfill{\spellLevel},
      colback=\cardContentColor,
      colframe=\cardBorderColor,
      coltitle=\cardTitleColor,
      reset and store to box array,
    ]
      \begin{tcolorbox}[
        tboxnospaces,
        boxrule=0pt,
        frame hidden, interior hidden,
        enhanced jigsaw,
      ]
        \vspace{-\baselineskip+0.6mm}
        \noindent\hspace*{1mm}
        \begin{tabularx}{\dimexpr\linewidth-2mm\relax}{@{}X X@{}}
          \SpellProperty{card/icons/clock-4.pdf}{\spellCastTime} &
          \SpellProperty{card/icons/timer.pdf}{
            \SpellDuration{\spellDuration}
          } \\[-8pt]
          \SpellProperty{card/icons/components.pdf}{%
            \SpellComponents{\spellVerbal}{\spellSomatic}{\spellMaterial}%
          } &
          \SpellProperty{card/icons/land-plot.pdf}{\spellDistance}
        \end{tabularx}
        \hspace*{1mm}%
        \vspace{-10pt}
      \end{tcolorbox}
      \vspace{1.2mm}
      \MaterialComponentsText{\spellMaterial}
      \SpellDescrition{\spellDescription}
    \end{tcolorbox}

    \ifboxarrayempty{2}{
      \begin{tcolorbox}[
        spellcard,
        colback=\cardContentColor,
        colframe=\cardBorderColor,
        store to box array,
      ]
        \begin{tcolorbox}[
          tboxnospaces,
          boxrule=0pt,
          opacityframe=1,opacityback=0,
          enhanced jigsaw,
        ]
        \end{tcolorbox}
      \end{tcolorbox}
    }

    \boxarraygetsize{\sidesCount}
    \foreach \n in {1,...,\sidesCount} {
      \usetcboxarray{\n}{
        on line, blankest,
      }
    }
}
